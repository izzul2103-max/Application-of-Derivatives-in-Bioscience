    <script>
        // 1. Core Mathematical Modeling Calculations
        // Function f(x) = 8 / (1 + 5 * e^(-0.5x))
        function calculateCanopyWidth(x) {
            return 8 / (1 + 5 * Math.exp(-0.5 * x));
        }

        // Derivative f'(x) = (20 * e^(-0.5x)) / (1 + 5 * Math.exp(-0.5x))^2
        function calculateGrowthRate(x) {
            const numerator = 20 * Math.exp(-0.5 * x);
            const denominator = Math.pow(1 + 5 * Math.exp(-0.5 * x), 2);
            return numerator / denominator;
        }

        // Generate data arrays for age 0 to 15 years
        const labels = [];
        const canopyData = [];
        const rateData = [];

        for (let i = 0; i <= 15; i += 0.5) {
            labels.push(i);
            canopyData.push(calculateCanopyWidth(i).toFixed(2));
            rateData.push(calculateGrowthRate(i).toFixed(2));
        }

        // 2. Layman Biological Translator Generator
        function getLaymanExplanation(age, rate, width) {
            if (age < 3) {
                return `At ${age} years old, the young palm is in its vegetative establishment phase. The canopy is expanding rapidly at a speed of ${rate} meters per year. Plantation operators should ensure optimal weed management around the base, as the tree is working hard to claim its space and maximize solar absorption.`;
            } else if (age >= 3 && age < 6) {
                return `At ${age} years, the palm reaches peak expansion velocity (around 4-5 years). The derivative value demonstrates high metabolic activity. The changing structure means the fronds are rapidly approaching interlocking territory with neighboring palms. Fertilizer application during this window directly translates to structural canopy readiness for early harvesting yield.`;
            } else if (age >= 6 && age < 10) {
                return `At ${age} years old, the tree has reached a broad size of ${width} meters, but notice the derivative rate is slowing down to ${rate} meters per year. The canopy expansion is plateauing because neighboring trees are meeting, creating light competition. The biological focus shifts from growing bigger fronds to sustaining optimal fruit bunch production.`;
            } else {
                return `At a mature stage of ${age} years, the expansion speed has dropped significantly down to ${rate} meters per year, meaning structural dimensions are stable. The canopy system has reached full maturity equilibrium. Further maintenance is focused on pruning lower dead fronds to facilitate harvesting visibility rather than managing space constraints.`;
            }
        }

        // 3. Initialize Interactive Chart.js Object
        const ctx = document.getElementById('growthChart').getContext('2d');
        const growthChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [
                    {
                        label: 'Canopy Diameter f(x) [Meters]',
                        data: canopyData,
                        borderColor: '#2563eb', // blue-600
                        backgroundColor: 'rgba(37, 99, 235, 0.05)',
                        borderWidth: 3,
                        pointRadius: 2,
                        pointHoverRadius: 6,
                        tension: 0.2,
                        yAxisID: 'y'
                    },
                    {
                        label: "Growth Velocity f'(x) [Meters/Year]",
                        data: rateData,
                        borderColor: '#10b981', // emerald-500
                        borderDash: [5, 5],
                        backgroundColor: 'transparent',
                        borderWidth: 2,
                        pointRadius: 0,
                        tension: 0.2,
                        yAxisID: 'y1'
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: {
                    duration: 2200,
                    easing: 'easeOutQuart'
                },
                interaction: {
                    mode: 'index',
                    intersect: false
                },
                plugins: {
                    legend: {
                        display: false // Turned off standard legends to favor tailored absolute badges
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                let label = context.dataset.label || '';
                                if (label) {
                                    label += ': ';
                                }
                                if (context.parsed.y !== null) {
                                    label += context.parsed.y;
                                }
                                return label;
                            }
                        }
                    }
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Palm Tree Age (Years Post-Planting)',
                            color: '#475569',
                            font: { weight: 'bold', size: 12 }
                        },
                        grid: { color: '#f1f5f9' }
                    },
                    y: {
                        type: 'linear',
                        display: true,
                        position: 'left',
                        title: {
                            display: true,
                            text: 'Canopy Width f(x) (Meters)',
                            color: '#2563eb',
                            font: { weight: 'bold' }
                        },
                        min: 0,
                        max: 10,
                        grid: { color: '#e2e8f0' }
                    },
                    y1: {
                        type: 'linear',
                        display: true,
                        position: 'right',
                        title: {
                            display: true,
                            text: "Rate of Change f'(x) (m/year)",
                            color: '#10b981',
                            font: { weight: 'bold' }
                        },
                        min: 0,
                        max: 2,
                        grid: { drawOnChartArea: false } // Avoid grid clutter
                    }
                },
                // Capture element clicks directly from user interactions
                onClick: (e) => {
                    const points = growthChart.getElementsAtEventForMode(e, 'index', { intersect: false }, true);
                    if (points.length) {
                        const firstPoint = points[0];
                        const index = firstPoint.index;
                        
                        const selectedAge = growthChart.data.labels[index];
                        const widthVal = growthChart.data.datasets[0].data[index];
                        const rateVal = growthChart.data.datasets[1].data[index];
                        
                        // Execute UI Engine panel updates
                        document.getElementById('consoleInstruction').classList.add('hidden');
                        
                        const dataPanel = document.getElementById('consoleData');
                        dataPanel.classList.remove('hidden');
                        
                        // Populate variable strings dynamically
                        document.getElementById('outAge').innerText = `${selectedAge} Years`;
                        document.getElementById('outWidth').innerText = `${widthVal} m`;
                        document.getElementById('outRate').innerText = `${rateVal} m/year`;
                        document.getElementById('outExpl').innerText = getLaymanExplanation(selectedAge, rateVal, widthVal);
                        
                        // Visual flash highlight feedback effect to signify updates
                        const consoleWidget = document.getElementById('resultConsole');
                        consoleWidget.classList.remove('bg-amber-50', 'border-amber-200');
                        consoleWidget.classList.add('bg-emerald-50', 'border-emerald-300');
                    }
                }
            }
        });
